## 一、复杂度、递归与 C++ 基础

### 1.1 时间复杂度与空间复杂度

常见复杂度（从小到大）：

$$O(1) < O(\log n) < O(n) < O(n\log n) < O(n^2) < O(2^n) < O(n!)$$

**分析规则：**

- 只看最高阶项
- 循环折半 → $O(\log n)$
- 嵌套循环 → 乘积
- 递归 → 递归树 / 主定理

### 1.2 递归复杂度与主定理

对于 $T(n) = aT(n/b) + O(n^d)$：

- 若 $d < \log_b a$，则 $T(n) = O(n^{\log_b a})$
- 若 $d = \log_b a$，则 $T(n) = O(n^d \log n)$
- 若 $d > \log_b a$，则 $T(n) = O(n^d)$

**示例：归并排序**

```cpp
// T(n) = 2T(n/2) + O(n) → O(n log n)
void mergeSort(vector<int>& a, int l, int r) {
    if (l >= r) return;
    int mid = (l + r) / 2;
    mergeSort(a, l, mid);
    mergeSort(a, mid + 1, r);
    merge(a, l, mid, r);
}
```

```cpp
    int m = matrix.size(), n = matrix[0].size();
    int top = 0, bottom = m - 1, left = 0, right = n - 1;
    vector<int> res;
    while (top <= bottom && left <= right) {
        for (int j = left; j <= right; j++) res.push_back(matrix[top][j]);
        top++;
        for (int i = top; i <= bottom; i++) res.push_back(matrix[i][right]);

            bottom--;
        }
        if (left <= right) {

            left++;
        }
    }
```

```cpp
    int pivot = a[r], i = l - 1;
```

### 3.2 归并排序

```cpp
void mergeSort(vector<int>& a, int l, int r) {
    if (l >= r) return;
    int mid = (l + r) / 2;
    mergeSort(a, l, mid);
    mergeSort(a, mid + 1, r);
    vector<int> tmp;
    int i = l, j = mid + 1;
    while (i <= mid && j <= r)
        tmp.push_back(a[i] < a[j] ? a[i++] : a[j++]);
    while (i <= mid) tmp.push_back(a[i++]);
    while (j <= r) tmp.push_back(a[j++]);
    for (int k = l; k <= r; k++) a[k] = tmp[k - l];
}
```

### 3.3 快速选择（第 K 大元素）

```cpp
int quickSelect(vector<int>& a, int l, int r, int k) {
    if (l == r) return a[l];
    int pivot = a[l + rand() % (r - l + 1)];
    int i = l, j = r;
    while (i <= j) {
        while (a[i] < pivot) i++;
        while (a[j] > pivot) j--;
        if (i <= j) swap(a[i++], a[j--]);
    }
    if (k <= j) return quickSelect(a, l, j, k);
    if (k >= i) return quickSelect(a, i, r, k);
    return a[k];
}
```

```cpp
int lengthOfLongestSubstring(string s) {
    unordered_map<char, int> cnt;
    int ans = 0;
    for (int l = 0, r = 0; r < s.size(); r++) {
        cnt[s[r]]++;
```

### 3.12 滑动窗口（最小覆盖子串）

```cpp
string minWindow(string s, string t) {
    unordered_map<char, int> need, window;
    for (char c : t) need[c]++;
    int valid = 0, start = 0, len = INT_MAX;
    for (int l = 0, r = 0; r < s.size(); r++) {
        char c = s[r];
        if (need.count(c)) {
            window[c]++;
            if (window[c] == need[c]) valid++;
        }
        while (valid == need.size()) {
            if (r - l + 1 < len) { start = l; len = r - l + 1; }
            char d = s[l++];
            if (need.count(d)) {
                if (window[d] == need[d]) valid--;
                window[d]--;
            }
        }
    }
    return len == INT_MAX ? "" : s.substr(start, len);
}
```

```cpp
class MedianFinder {
    priority_queue<int> left;  // 大根堆，存较小的一半
    priority_queue<int, vector<int>, greater<int>> right;  // 小根堆
public:
    void addNum(int num) {
        if (left.empty() || num <= left.top()) left.push(num);
        else right.push(num);
        if (left.size() > right.size() + 1) { right.push(left.top()); left.pop(); }
        else if (right.size() > left.size()) { left.push(right.top()); right.pop(); }
    }
    double findMedian() {
        if (left.size() > right.size()) return left.top();
        return (left.top() + right.top()) / 2.0;
    }
};
```

### 5.8 LRU 缓存（哈希表 + 双向链表）

```cpp
class LRUCache {
    int cap;
    list<pair<int,int>> cache;  // (key, value)
    unordered_map<int, list<pair<int,int>>::iterator> mp;
public:
    LRUCache(int capacity) : cap(capacity) {}
    int get(int key) {
        if (!mp.count(key)) return -1;
        cache.splice(cache.begin(), cache, mp[key]);
        return mp[key]->second;
    }
    void put(int key, int value) {
        if (mp.count(key)) {
            cache.splice(cache.begin(), cache, mp[key]);
            mp[key]->second = value;
        } else {
            if (cache.size() == cap) {
                mp.erase(cache.back().first);
                cache.pop_back();
            }
            cache.push_front({key, value});
            mp[key] = cache.begin();
        }
    }
};
```

### 7.7 Dijkstra（堆优化）

```cpp
vector<int> dijkstra(int start, vector<vector<pair<int,int>>>& adj) {
    int n = adj.size();
    vector<int> dist(n, INT_MAX);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq;
    dist[start] = 0; pq.push({0, start});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (auto [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

```cpp
struct Edge { int u, v, w; };
int bellmanFord(vector<Edge>& edges, int n, int start) {
    vector<int> dist(n, INT_MAX);
    dist[start] = 0;
    for (int i = 0; i < n - 1; i++)
        for (auto& e : edges)
            if (dist[e.u] != INT_MAX)
                dist[e.v] = min(dist[e.v], dist[e.u] + e.w);
    // 第 n 轮检查负环
    for (auto& e : edges)
        if (dist[e.u] != INT_MAX && dist[e.u] + e.w < dist[e.v])
            return -1;  // 存在负环
    return 0;
}
```

### 7.12 并查集

```cpp
struct DSU {
    vector<int> fa, sz;
    DSU(int n) : fa(n), sz(n, 1) { iota(fa.begin(), fa.end(), 0); }
    int find(int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); }
    bool unite(int x, int y) {
        x = find(x); y = find(y);
        if (x == y) return false;
        if (sz[x] < sz[y]) swap(x, y);
        fa[y] = x; sz[x] += sz[y];
        return true;
    }
    bool same(int x, int y) { return find(x) == find(y); }
};
```

```cpp
int dfs(int i, int j) {
    if (i < 0 || j < 0) return 0;
    if (memo[i][j] != -1) return memo[i][j];
    return memo[i][j] = max(dfs(i-1, j), dfs(i, j-1)) + grid[i][j];
}
```

```cpp
    int m = p.size();
```

### 11.2 KMP 匹配

```cpp
int kmp(string& s, string& p) {
    auto nxt = getNext(p);
    int n = s.size(), m = p.size();
    for (int i = 0, j = 0; i < n; i++) {
        while (j > 0 && s[i] != p[j]) j = nxt[j - 1];
        if (s[i] == p[j]) j++;
        if (j == m) return i - m + 1;  // 匹配成功
    }
    return -1;
}
```

```cpp
    vector<vector<bool>> dp(n, vector<bool>(n, false));
    for (int i = n - 1; i >= 0; i--)
        for (int j = i; j < n; j++)
```

### 12.9 快速幂

```cpp
long long qpow(long long a, long long b, long long mod) {
    long long res = 1;
    while (b) {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}
```

### 12.10 矩阵快速幂

```cpp
typedef vector<vector<long long>> Matrix;
Matrix mul(Matrix& A, Matrix& B, long long mod) {
    int n = A.size();
    Matrix C(n, vector<long long>(n, 0));
    for (int i = 0; i < n; i++)
        for (int k = 0; k < n; k++)
            for (int j = 0; j < n; j++)
                C[i][j] = (C[i][j] + A[i][k] * B[k][j]) % mod;
    return C;
}
Matrix matPow(Matrix A, long long b, long long mod) {
    int n = A.size();
    Matrix res(n, vector<long long>(n, 0));
    for (int i = 0; i < n; i++) res[i][i] = 1;
    while (b) {
        if (b & 1) res = mul(res, A, mod);
        A = mul(A, A, mod);
        b >>= 1;
    }
    return res;
}
```

### 12.14 容斥原理

```cpp
// 求 [1, n] 中能被集合 A 中至少一个数整除的个数
long long inclusionExclusion(int n, vector<int>& A) {
    long long ans = 0;
    int m = A.size();
    for (int mask = 1; mask < (1 << m); mask++) {
        long long lcm = 1;
        int bits = 0;
        for (int i = 0; i < m; i++)
            if (mask & (1 << i)) {
                bits++;
                lcm = lcm / gcd(lcm, A[i]) * A[i];
                if (lcm > n) break;
            }
        if (bits % 2 == 1) ans += n / lcm;
        else ans -= n / lcm;
    }
    return ans;
}
```

### 12.15 高斯消元

```cpp
vector<double> gauss(vector<vector<double>>& a) {
    int n = a.size();
    for (int i = 0; i < n; i++) {
        int r = i;
        for (int j = i + 1; j < n; j++)
            if (fabs(a[j][i]) > fabs(a[r][i])) r = j;
        swap(a[i], a[r]);
        for (int j = i + 1; j <= n; j++) a[i][j] /= a[i][i];
        a[i][i] = 1;
        for (int k = 0; k < n; k++)
            if (k != i)
                for (int j = i + 1; j <= n; j++)
                    a[k][j] -= a[k][i] * a[i][j];
    }
    vector<double> ans(n);
    for (int i = 0; i < n; i++) ans[i] = a[i][n];
    return ans;
}
```
